cmake_minimum_required(VERSION 3.22)
project(reader LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(USE_ENCRYPTION OFF CACHE BOOL "" FORCE)
set(USE_LIBXML2 OFF CACHE BOOL "" FORCE)
set(USE_XMLWRITER ON CACHE BOOL "" FORCE)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../..)
add_subdirectory(${ROOT}/vendor/libmobi ${CMAKE_CURRENT_BINARY_DIR}/mobi EXCLUDE_FROM_ALL)
add_library(reader SHARED reader.cpp)
target_include_directories(reader PRIVATE ${ROOT}/vendor/libmobi/src)
target_link_libraries(reader mobi android log)
target_compile_options(reader PRIVATE -O3)
